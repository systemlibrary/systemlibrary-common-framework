SystemLibrary.Common.Framework.Extensions

EnumExtensions<TEnum>

Static methods for enum types — use when extension methods on generic types are needed.

class EnumExtensions<TEnum>

{

Returns all enum values as TEnum.

static IEnumerable<TEnum> GetEnums()

✕ 
var enums = EnumExtensions<Color>.GetEnums();
// { Color.Black, Color.White }

Returns all enum key names as strings, ignoring any EnumText or EnumValue attributes.

static IEnumerable<string> GetKeys()

✕ 
enum Color { [EnumText("Black Colored Text")] Black, White }

var keys = EnumExtensions<Color>.GetKeys();
// { "Black", "White" }

Returns all enum values via ToValue() — respects EnumValue attributes, falls back to key name.

static IEnumerable<string> GetValues()

✕ 
enum Color { [EnumValue("#000")] Black, White }

var values = EnumExtensions<Color>.GetValues();
// { "#000", "White" }
}
